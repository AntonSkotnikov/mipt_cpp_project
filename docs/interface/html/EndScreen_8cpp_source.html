9bool humanityWon(const GameSnapshot & snapshot) {
10 if (snapshot.cureProgress >= 100.0) {
11 return true;
12 }
13
14 if (snapshot.countries.empty()) {
15 return false;
16 }
17
18 double living = 0.0;
19 double infected = 0.0;
20 for (const Country & country : snapshot.countries) {
21 living += country.pop.susceptible + country.pop.exposed + country.pop.infected + country.pop.recovered;
22 infected += country.pop.exposed + country.pop.infected;
23 }
24
25 return living > 0.0 && infected <= 0.0;
26}

56void EndScreen::updateText() {
57 if (resultInfo_ == nullptr) {
58 return;
59 }
60
61 const bool humanityVictory = humanityWon(snapshot_);
62 const bool playerVictory =
63 (snapshot_.playerInfo.role == PlayerRole::Humanity && humanityVictory) ||
64 (snapshot_.playerInfo.role == PlayerRole::Pathogen && !humanityVictory);
65
66 std::ostringstream out;
67 out << (playerVictory ? "Victory" : "Defeat") << "\n\n";
68 out << "Winner: " << (humanityVictory ? "Humanity" : "Pathogen") << "\n";
69 out << "Your side: " << roleTitle(snapshot_.playerInfo.role) << "\n";
70 out << "Day: " << snapshot_.day << "\n";
71 out << "Cure progress: " << static_cast<int>(snapshot_.cureProgress) << "%\n\n";
72
73 if (humanityVictory) {
74 out << "The outbreak has been contained. Humanity survives.";
75 } else {
76 out << "The pathogen overwhelmed the world. Humanity has fallen.";
77 }
78
79 resultInfo_->changeText(out.str());
80}

82void EndScreen::layout() {
83 if (widgets.size() < 2) return;
84
85 const int padding = win_.bordered() ? 2 : 1;
86 const int contentWidth = std::max(1, win_.width() - padding * 2);
87 const int contentHeight = std::max(1, win_.height() - padding * 2);
88 const int panelWidth = std::min(72, std::max(40, contentWidth / 2));
89 const int panelHeight = std::min(14, std::max(8, contentHeight - 8));
90 const int panelX = padding + std::max(0, (contentWidth - panelWidth) / 2);
91 const int panelY = padding + std::max(0, (contentHeight - panelHeight - 4) / 2);
92 const int buttonWidth = std::min(24, std::max(14, panelWidth / 3));
93 const int buttonX = padding + std::max(0, (contentWidth - buttonWidth) / 2);
94 const int buttonY = panelY + panelHeight + 2;
95
96 widgets[0]->setRect({panelY + 1, panelX + 1, panelHeight - 2, panelWidth - 2});
97 widgets[1]->setRect({buttonY + 1, buttonX + 1, 1, buttonWidth - 2});
98}
