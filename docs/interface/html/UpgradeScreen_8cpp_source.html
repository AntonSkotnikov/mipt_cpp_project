38void UpgradeScreen::layout() {
39 if (widgets.size() <= descriptionIndex_) return;
40
41 const int padding = win_.bordered() ? 2 : 1;
42 const int contentX = padding;
43 const int contentY = padding;
44 const int contentWidth = std::max(1, win_.width() - padding * 2);
45 const int contentHeight = std::max(1, win_.height() - padding * 2);
46 const int gap = 1;
47 const int tabHeight = 3;
48 const int tabWidth = std::max(12, (contentWidth - gap * static_cast<int>(tabCount_ - 1)) / static_cast<int>(tabCount_));
49
50 for (std::size_t i = 0; i < tabCount_; i++) {
51 const int x = contentX + static_cast<int>(i) * (tabWidth + gap);
52 const int width = i == tabCount_ - 1
53 ? std::max(1, contentX + contentWidth - x)
54 : tabWidth;
55 widgets[i]->setRect(innerRect({contentY, x, tabHeight, width}));
56 }
57
58 const int bodyY = contentY + tabHeight + gap + 1;
59 const int bodyHeight = std::max(1, contentHeight - tabHeight - gap - 2);
60 const int listWidth = std::min(42, std::max(28, contentWidth / 3));
61 const int descriptionX = contentX + listWidth + gap;
62 const int descriptionWidth = std::max(1, contentX + contentWidth - descriptionX);
63
64 widgets[listIndex_]->setRect({
65 bodyY + 1,
66 contentX,
67 std::max(1, bodyHeight - 1),
68 std::max(1, listWidth - 1)
69 });
70 widgets[descriptionIndex_]->setRect({
71 bodyY + 1,
72 descriptionX,
73 std::max(1, bodyHeight - 1),
74 descriptionWidth
75 });
76}

78void UpgradeScreen::updateDescription() {
79 if (upgradeList_ == nullptr || description_ == nullptr) {
80 return;
81 }
82
83 const UpgradeListItem * selected = upgradeList_->selectedItem();
84 if (selected == nullptr) {
85 description_->changeText("No upgrades available");
86 return;
87 }
88
89 std::string dependencies = "None";
90 if (!selected->upgrade.dependencies.empty()) {
91 dependencies.clear();
92 for (std::size_t i = 0; i < selected->upgrade.dependencies.size(); i++) {
93 if (i > 0) {
94 dependencies += ", ";
95 }
96 dependencies += selected->upgrade.dependencies[i];
97 }
98 }
99
100 std::string status = "Available";
101 if (selected->purchased) {
102 status = "Purchased";
103 } else if (!selected->available) {
104 status = "Locked";
105 }
106
107 description_->changeText(
108 selected->upgrade.title +
109 "\n\nCost: " + std::to_string(selected->upgrade.cost) +
110 "\nStatus: " + status +
111 "\nDependencies: " + dependencies +
112 "\n\n" + selected->upgrade.description
113 );
114}

160request::UIRequest UpgradeScreen::handleInput(int key) {
161 if (key == 27) {
162 return request::Game::Back;
163 }
164
165 if ((key == KEY_ENTER || key == '\n' || key == '\r') && focusedIndex_ == listIndex_) {
166 return purchaseSelectedUpgrade();
167 }
168
169 const InputResult result = handleFocusedInput(key);
170 if (!std::holds_alternative<request::None>(result.request)) {
171 return result.request;
172 }
173
174 if (result.handled) {
175 updateDescription();
176 return request::None{};
177 }
178
179 switch (key) {
180 case KEY_UP:
181 focusWidget(listIndex_);
182 updateDescription();
183 return request::None{};
184
185 case KEY_DOWN:
186 focusWidget(listIndex_);
187 updateDescription();
188 return request::None{};
189
190 case KEY_LEFT:
191 if (focusedIndex_ < tabCount_) {
192 focusPrevTab();
193 } else {
194 focusTab(upgradeTabIndex(category_));
195 }
196 return request::None{};
197
198 case KEY_RIGHT:
199 if (focusedIndex_ < tabCount_) {
200 focusNextTab();
201 } else {
202 focusTab(upgradeTabIndex(category_));
203 }
204 return request::None{};
205
206 case KEY_BTAB:
207 focusPrev();
208 updateDescription();
209 return request::None{};
210
211 case '\t':
212 focusNext();
213 updateDescription();
214 return request::None{};
215
216 case 't':
217 case 'T':
218 return request::Game::Transmission;
219
220 case 'c':
221 case 'C':
222 return request::Game::Clinic;
223
224 case 'a':
225 case 'A':
226 return request::Game::Abilities;
227
228 case 'u':
229 case 'U':
230 return upgradeTabRequests[upgradeTabIndex(category_)];
231 }
232
233 return request::None{};
234}
