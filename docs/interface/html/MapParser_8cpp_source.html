58std::size_t utf8SymbolLength(unsigned char ch) {
59 if ((ch & 0b10000000) == 0) return 1;
60 if ((ch & 0b11100000) == 0b11000000) return 2;
61 if ((ch & 0b11110000) == 0b11100000) return 3;
62 if ((ch & 0b11111000) == 0b11110000) return 4;
63 return 1;
64}

76std::vector<SymbolOnScreen> parseMapFile(const std::filesystem::path & path) {
77 std::ifstream input(path);
78 if (!input) {
79 throw std::runtime_error("Could not open map file: " + path.string());
80 }
81
82 std::vector<SymbolOnScreen> symbols;
83 std::string line;
84 int y = 0;
85
86 while (std::getline(input, line)) {
87 int x = 0;
88
89 for (std::size_t i = 0; i < line.size();) {
90 const unsigned char ch = static_cast<unsigned char>(line[i]);
91 const std::size_t length = std::min(utf8SymbolLength(ch), line.size() - i);
92 std::string symbol = line.substr(i, length);
93
94 if (symbol != " " && symbol != "\t" && symbol != "\r") {
95 symbols.push_back({y, x, std::move(symbol)});
96 }
97
98 i += length;
99 x++;
100 }
101
102 y++;
103 }
104
105 return symbols;
106}

112std::vector<SymbolOnScreen> parseMapCountry(const std::string & countryName, Resolutions resolution) {
113 const MapDirSet dirs = dirsForResolution(resolution);
114 const auto candidates = mapFileNameCandidates(countryName);
115
116 for (std::size_t i = 0; i < dirs.count; i++) {
117 for (std::string_view candidate : candidates) {
118 const std::filesystem::path path = std::filesystem::path(dirs.dirs[i]) / candidate;
119 if (std::filesystem::exists(path)) {
120 return parseMapFile(path);
121 }
122 }
123 }
124
125 return parseMapFile(std::filesystem::path(dirs.dirs[0]) / countryName);
126}
