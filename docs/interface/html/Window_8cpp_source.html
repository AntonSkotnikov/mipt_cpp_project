86void Window::printCentered(int y, std::string_view text) {
87 int height = 0;
88 int width = 0;
89 getmaxyx(win_, height, width);
90
91 if (y >= height || width <= 0) return;
92
93 std::vector<std::string> lines;
94 std::string currentLine;
95
96 const char * ptr = text.data();
97
98 while (*ptr) {
99 if (*ptr == '\n') {
100 lines.push_back(currentLine);
101 currentLine.clear();
102 ++ptr;
103 continue;
104 }
105
106 currentLine.push_back(*ptr);
107
108 if ((int)currentLine.size() >= width) {
109 lines.push_back(currentLine);
110 currentLine.clear();
111 }
112
113 ++ptr;
114 }
115
116 if (!currentLine.empty()) {
117 lines.push_back(currentLine);
118 }
119
120 for (size_t i = 0; i < lines.size(); ++i) {
121 int currentY = y + static_cast<int>(i);
122 if (currentY >= height) break;
123
124 const std::string& line = lines[i];
125 int lineLen = static_cast<int>(line.size());
126
127 int x = (width - lineLen) / 2;
128 if (x < 0) x = 0;
129
130 mvwprintw(win_, currentY, x, "%s", line.c_str());
131 }
