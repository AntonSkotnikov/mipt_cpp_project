78void ScreenManager::applyWindowLayout() {
79 int terminalHeight = 0;
80 int terminalWidth = 0;
81 getmaxyx(stdscr, terminalHeight, terminalWidth);
82
83 if (cfg_.terminalTooSmall) {
84 mainWin_.setBorders(false);
85 mainWin_.resizeCentered(terminalHeight, terminalWidth);
86 return;
87 }
88
89 const TerminalProfile profile = terminalProfiles.at(cfg_.resolution);
90 const bool profileMatchesTerminal = terminalHeight == profile.height && terminalWidth == profile.width;
91 const bool bordered = !profileMatchesTerminal;
92 mainWin_.setBorders(bordered);
93
94 const int targetHeight = profile.height + (bordered ? deltaForBorders : 0);
95 const int targetWidth = profile.width + (bordered ? deltaForBorders : 0);
96 mainWin_.resizeCentered(targetHeight, targetWidth);
97}
