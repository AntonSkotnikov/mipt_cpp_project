44std::vector<const Country *> topMedicineCountries(const std::vector<Country> & countries) {
45 std::vector<const Country *> result;
46 result.reserve(countries.size());
47 for (const Country & country : countries) {
48 result.push_back(&country);
49 }
50
51 std::sort(result.begin(), result.end(), [](const Country * lhs, const Country * rhs) {
52 const CountryParams lhsParams = countryParamsForName(lhs->name);
53 const CountryParams rhsParams = countryParamsForName(rhs->name);
54 if (lhsParams.medicine != rhsParams.medicine) {
55 return lhsParams.medicine > rhsParams.medicine;
56 }
57 return lhsParams.governmentReaction > rhsParams.governmentReaction;
58 });
59
60 if (result.size() > 5) {
61 result.resize(5);
62 }
63 return result;
64}
